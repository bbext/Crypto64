MODULE CryptoTestCiphers;

	IMPORT U := CryptoUtils, Ciphers := CryptoCiphers, Kernel, Log := StdLog, AO := CryptoAosCompat;

	PROCEDURE Ecb1* (modname: ARRAY OF CHAR; keybits: AO.SIGNED32);
		VAR
			hex, bindata, binkey, orig: ARRAY 64 + 1 OF SHORTCHAR;
			cipher: Ciphers.Cipher;
	BEGIN
	   (*     c.arg.SkipWhitespace;
		c.arg.String(modname);
		c.arg.SkipWhitespace;
		c.arg.Int(keybits, FALSE);*)
		cipher := Ciphers.NewCipher(SHORT(modname));
		hex := "0123456789ABCDEF0123456789ABCDEF0123456789ABCDEF0123456789ABCDEF";
		U.Hex2Bin(hex, 0, binkey, 0, keybits DIV 8);
		U.Hex2Bin(hex, 0, bindata, 0, cipher.blockSize);
		orig := bindata;
		orig[cipher.blockSize] := 0X;
		cipher.InitKey(binkey, keybits);
		Log.Ln;
		Log.String("*********************************");
		Log.Ln;
		Log.String("Encrypt-Decrypt-Test in ECB-mode: ");
		Log.String(LONG(cipher.name));
		Log.Ln;
		Log.String("Key: ");
		U.PrintHex(binkey, 0, keybits DIV 8);
		Log.Ln;
		Log.String("Original: ");
		U.PrintHex(bindata, 0, cipher.blockSize);
		cipher.Encrypt(bindata, 0, cipher.blockSize);
		Log.Ln;
		Log.String("Encrypted: ");
		U.PrintHex(bindata, 0, cipher.blockSize);
		cipher.InitKey(binkey, keybits);
		cipher.Decrypt(bindata, 0, cipher.blockSize);
		Log.Ln;
		Log.String("Decrypted: ");
		U.PrintHex(bindata, 0, cipher.blockSize);
		bindata[cipher.blockSize] := 0X;
		Log.Ln;
		IF bindata = orig THEN
			Log.String("OK")
		ELSE
			Log.String("FAIL")
		END;
		Log.Ln
	END Ecb1;

	PROCEDURE Ecb2 (IN modname, input, output, key : ARRAY OF SHORTCHAR; datalen, keybits : AO.SIGNED32);
		VAR
			cipher: Ciphers.Cipher;
			temp1, temp2: ARRAY 256 OF SHORTCHAR;


	BEGIN
		cipher := Ciphers.NewCipher(modname);
		U.Hex2Bin(key, 0, temp1, 0, keybits DIV 8);
		cipher.InitKey(temp1, keybits);
		Log.Ln;
		Log.String("*********************************");
		Log.Ln;
		Log.String("Encryption-Test: ");
		Log.String(LONG(cipher.name));
		Log.Ln;
		Log.String("Key: ");
		U.PrintHex(temp1, 0, keybits DIV 8);
		U.Hex2Bin(input, 0, temp1, 0, datalen);
		Log.Ln;
		Log.String("plaintext: ");
		U.PrintHex(temp1, 0, datalen);
		cipher.Encrypt(temp1, 0, datalen);
		Log.Ln;
		Log.String("encryption: ");
		U.PrintHex(temp1, 0, datalen);
		U.Hex2Bin(output, 0, temp2, 0, datalen);
		Log.Ln;
		Log.String("correct encryption: ");
		U.PrintHex(temp2, 0, datalen);
		Log.Ln;
		temp1[datalen] := 0X;
		temp2[datalen] := 0X;
		IF temp1 = temp2 THEN
			Log.String("OK")
		ELSE
			Log.String("FAIL")
		END;
		Log.Ln
	END Ecb2;

	PROCEDURE MeasureTime* (modname, mode: ARRAY OF CHAR; keybits: AO.SIGNED32);
		VAR
			buf, key: ARRAY 1024 OF SHORTCHAR;
			i, j, k, t: AO.SIGNED32;
			cipher: Ciphers.Cipher;
			iv: ARRAY 64 OF SHORTCHAR;
						  t0: LONGINT; 

	BEGIN

		FOR i := 0 TO 1023 DO
			buf[i] := 'a'
		END;
		cipher := Ciphers.NewCipher(SHORT(modname));
		cipher.InitKey(key, keybits);
		IF mode = "CBC" THEN
			cipher.SetIV(iv, Ciphers.CBC)
		END;
		Log.Ln;
		Log.String("***********************************");
		Log.Ln;
		Log.String("Encrypting 100 MB with ");
		Log.String(LONG(cipher.name));
		t0 := Kernel.Time();
		FOR k := 0 TO 9 DO
			Log.String(".");
			FOR j := 0 TO 9 DO
				FOR i := 0 TO 999 DO
					cipher.Encrypt(buf, 0, 1024)
				END
			END
		END;
				Log.Int(SHORT((Kernel.Time() - t0) * 1000 DIV Kernel.timeResolution));
		Log.String(" ms");
		Log.Ln
	END MeasureTime;

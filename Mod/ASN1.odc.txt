	IMPORT Streams := CryptoAosStreams, AO := CryptoAosCompat;

	CONST
		Trace = TRUE;
		ASN1Universal* = 0;
		ASN1Application* = 1;
		ASN1Context* = 2;
		ASN1Private* = 3;
		Boolean* = 1H;
		Integer* = 2H;
		BitString* = 3H;
		String* = 4H;
		Null* = 5H;
		Oid* = 6H;
		UTF8* = 0CH;
		PrintableString* = 13H;
		TeletexString* = 14H;
		IA5String* = 16H;
		UTCTime* = 17H;
		BMPString* = 1EH;
		Sequence* = 30H;
		Set* = 31H;

	TYPE
		CharString* = POINTER TO ARRAY OF SHORTCHAR;


	TYPE
		Triplet* = POINTER TO EXTENSIBLE RECORD
			tag* : AO.SIGNED32;
			class* : AO.SIGNED32;
			length* : AO.SIGNED32;
			unusedbits* : AO.SIGNED32;
			constructed* : BOOLEAN;
			definite* : BOOLEAN;
			bvalue* : BOOLEAN;
			ivalue* : AO.SIGNED32;
			svalue* : POINTER TO ARRAY OF SHORTCHAR;
			child*, curchild* : Triplet;
			next* : Triplet;
			level* : AO.SIGNED32
		END;


	VAR
		log: Streams.Writer;

	PROCEDURE (SELF: Triplet) AppendChild* (t : Triplet), NEW;

	BEGIN
		IF SELF.child = NIL THEN
			SELF.child := t;
			SELF.curchild := t
		ELSE
			SELF.curchild.next := t;
			SELF.curchild := t
		END
	END AppendChild;


	PROCEDURE PrintHexString* (w : Streams.Writer; IN s : ARRAY OF SHORTCHAR);
		VAR i: AO.SIZE; c: SHORTCHAR; hex : ARRAY 17 OF SHORTCHAR;
	BEGIN
		hex := "0123456789ABCDEF";
		FOR i := 0 TO LEN(s) - 1 DO
			c := s[i];
			w.Char(hex[ORD(c) DIV 16]);
			w.Char(hex[ORD(c) MOD 16]);
			w.Char(" ")
		END
	END PrintHexString;

	PROCEDURE PrintNumericString* (w : Streams.Writer; IN s : ARRAY OF SHORTCHAR);
		VAR i: AO.SIZE; c: SHORTCHAR;
	BEGIN
		w.String("  ");
		FOR i := 0 TO LEN(s) - 1 DO
			c := s[i];
			w.Int(ORD(c), 0);
			w.Char(".")
		END
	END PrintNumericString;

	PROCEDURE PrintString* (w : Streams.Writer; IN s : ARRAY OF SHORTCHAR);
		VAR i: AO.SIZE; c: SHORTCHAR;
	BEGIN
		FOR i := 0 TO LEN(s) - 1 DO
			c := s[i];
			IF (ORD(c) >= 20H) & (ORD(c) <= 7EH) THEN
				w.Char(c)
			ELSE
				w.Char(".")
			END
		END
	END PrintString;

	PROCEDURE (SELF: Triplet) Print* (w : Streams.Writer), NEW;
		VAR
			i: AO.SIGNED32;
	BEGIN
		w.Char(09X);
		FOR i := 0 TO SELF.level - 1 DO
			w.String("- ")
		END;
		w.Hex(SELF.tag, 4);
		w.String("H[");
		w.Int(SELF.length, 0);
		w.String("]");
		CASE SELF.tag OF
			Boolean :	       IF SELF.bvalue THEN
			w.String("true")
		ELSE
			w.String("false")
		END
		|       Integer :	       IF SELF.length <= 4 THEN
			w.Int(SELF.ivalue, 0)
		ELSE
			PrintHexString(w, SELF.svalue^)
		END
		|       Oid :	   PrintNumericString(w, SELF.svalue^);
		w.String("  ");
		PrintHexString(w, SELF.svalue^)
		|       PrintableString :	       PrintString(w, SELF.svalue^)
		|       UTCTime :	       PrintString(w, SELF.svalue^)
		|       BitString :	     w.Char("(");
		w.Int(SELF.unusedbits, 0);
		w.Char(")");
		IF SELF.svalue # NIL THEN
			PrintHexString(w, SELF.svalue^)
		END
		|       Sequence,Set :
		ELSE
			IF SELF.svalue # NIL THEN
				PrintHexString(w, SELF.svalue^)
			END
		END;
		w.Ln
	END Print;

	PROCEDURE Decode* (reader : Streams.Reader; level : AO.SIGNED32; VAR len : AO.SIZE): Triplet;
		VAR
			t, t1: Triplet;
			lengthbytes, len0, hdrlen, i: AO.SIZE;
			c: SHORTCHAR;


	BEGIN
		NEW(t);
		t.level := level;
		len := 0;
		hdrlen := 0;
		reader.Char(c);
		INC(len);
		INC(hdrlen);
		IF Trace & (log # NIL) THEN
			log.Char("{");
			log.Hex(ORD(c), 0);
			log.Char("H");
			log.Char("}")
		END;
		t.class := ORD(c) DIV 64;
		t.constructed := ODD(ORD(c) DIV 32);
		IF ORD(c) MOD 32 # 31 THEN
			t.tag := ORD(c) MOD 64
		ELSE
			REPEAT
				reader.Char(c);
				INC(len);
				INC(hdrlen);
				t.tag := t.tag * 128 + ORD(c) MOD 128
			UNTIL ORD(c) DIV 128 # 1
		END;
		reader.Char(c);
		INC(len);
		INC(hdrlen);
		IF ORD(c) < 128 THEN
			t.length := ORD(c);
			t.definite := TRUE
		ELSIF ORD(c) = 128 THEN
			t.definite := FALSE
		ELSE
			lengthbytes := ORD(c) MOD 128;
			t.definite := TRUE;
			FOR i := 0 TO lengthbytes - 1 DO
				t.length := 256 * t.length;
				reader.Char(c);
				INC(len);
				INC(hdrlen);
				t.length := t.length + ORD(c)
			END
		END;
		CASE t.tag OF
			Boolean :	       reader.Char(c);
		INC(len);
		t.bvalue := c # 0X
		|       Integer :	       IF t.length <= 4 THEN
			NEW(t.svalue, t.length);
			FOR i := 0 TO t.length - 1 DO
				t.ivalue := 256 * t.ivalue;
				reader.Char(c);
				INC(len);
				t.svalue[i] := c;
				t.ivalue := t.ivalue + ORD(c)
			END
		ELSE
			NEW(t.svalue, t.length);
			reader.Bytes(t.svalue^, 0, t.length, len0);
			len := len + len0
		END
		|       BitString :	     reader.Char(c);
		t.unusedbits := ORD(c);
		INC(len);
		IF FALSE & (ORD(reader.Peek()) = 30H) THEN
			IF Trace & (log # NIL) THEN
				log.String("(BitSRec  ");
				log.Int(len, 0);
				log.String(")");
				t.Print(log)
			END;
			t.child := Decode(reader, level + 1, len0);
			len := len + len0
		ELSE
			NEW(t.svalue, t.length - 1);
			IF t.length > 0 THEN
				reader.Bytes(t.svalue^, 0, t.length - 1, len0);
				len := len + len0
			END
		END
		|       String :		NEW(t.svalue, t.length);
		reader.Bytes(t.svalue^, 0, t.length, len0);
		len := len + len0
		|       Set :	   IF Trace & (log # NIL) THEN
			log.String("(Set  ");
			log.Int(len, 0);
			log.String("+)");
			t.Print(log)
		END;
		t.child := Decode(reader, level + 1, len0);
		len := len + len0
		|       Sequence :	      IF Trace & (log # NIL) THEN
			log.String("(Seq ");
			log.Int(len, 0);
			log.String("+)");
			t.Print(log)
		END;
		WHILE len - hdrlen < t.length DO
			t1 := Decode(reader, level + 1, len0);
			len := len + len0;
			t.AppendChild(t1)
		END
		ELSE
			
			IF t.length > 0 THEN
				NEW(t.svalue, t.length);
				reader.Bytes(t.svalue^, 0, t.length, len0);
				len := len + len0
			END
		END;
		IF Trace & (log # NIL) THEN
			CASE t.tag OF
				Integer :		       log.String("(Integer  ");
			log.Int(len, 4);
			log.Char(")");
			t.Print(log)
			|       BitString :		     IF t.constructed THEN
				log.String("(constrBitString ")
			ELSE
				log.String("(BitString ")
			END;
			log.Int(len, 4);
			log.Char(")");
			t.Print(log)
			|       String :			log.String("(String   ");
			log.Int(len, 4);
			log.Char(")");
			t.Print(log)
			|       Null :		  log.String("(Null     ");
			log.Int(len, 4);
			log.Char(")");
			t.Print(log)
			|       Oid :		   log.String("(OID");
			log.Int(len, 4);
			log.Char(")");
			t.Print(log)
			|       PrintableString :		       log.String("(Printable");
			log.Int(len, 4);
			log.Char(")");
			t.Print(log)
			|       UTCTime :		       log.String("(UTCTime");
			log.Int(len, 4);
			log.Char(")");
			t.Print(log)
			|       Set :
			|       Sequence :
			ELSE
				log.String("(t.tag=");
				log.Hex(t.tag, 4);
				log.Int(len, 4);
				log.Char(")");
				t.Print(log)
			END
		END;
		RETURN t
	END Decode;

BEGIN


END CryptoASN1.
